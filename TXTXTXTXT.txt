InstancePaths buildInstancePaths(
    const fs::path &instanceDirectory)
{
    InstancePaths paths;

    paths.operationPath =
        (instanceDirectory / "operations.csv").string();

    paths.jobPath =
        (instanceDirectory / "jobs.csv").string();

    paths.setupPath =
        (instanceDirectory / "setup.csv").string();

    paths.initialSetupPath =
        (instanceDirectory / "initial_setup.csv").string();

    return paths;
}

// ============================================================
// Verifica se uma pasta realmente contém uma instância
// ============================================================

bool isInstanceDirectory(
    const fs::path &directory)
{
    if (!fs::is_directory(directory))
        return false;

    return fs::exists(directory / "operations.csv") &&
           fs::exists(directory / "jobs.csv") &&
           fs::exists(directory / "setup.csv") &&
           fs::exists(directory / "initial_setup.csv");
}

// ============================================================
// Limite de tempo
// ============================================================

long getTimeLimit(
    const std::string &instanceName)
{
    (void)instanceName;
    return TIME_LIMIT_MS;
}

// ============================================================
// Parâmetros da Critical Path
// ============================================================

Parameters buildCriticalParameters(
    long timeLimitMs)
{
    Parameters parameters;

    // =====================================
    // Vizinhança Critical Path
    // =====================================

    parameters.neighborhood = 2;

    // Nos testes anteriores da Critical
    // estávamos utilizando tenure = 5.
    parameters.tabuTenure = 5;

    parameters.cycleWindow = 6;
    parameters.cycleRepeats = 3;

    parameters.checkpointStackSize = 20;

    // Deixamos muito alto para que o tempo
    // seja normalmente o critério dominante.
    parameters.iterationsWithoutImprovement =
        555000;

    parameters.maxMilliSeconds =
        timeLimitMs;

    return parameters;
}

// ============================================================
// Executa UMA instância
// ============================================================

double runCritical(
    const fs::path &instanceDirectory,
    double &bestMakespan,
    double &executionTime,
    long timeLimitMs)
{
    // Cada execução recebe uma Instance nova.
    // Assim uma instância não contamina a outra.
    Instance instance;

    InstancePaths paths =
        buildInstancePaths(
            instanceDirectory);

    instance.initializeInstance(paths);

    int machineCount =
        static_cast<int>(
            instance.machOperation.size());

    Parameters parameters =
        buildCriticalParameters(
            timeLimitMs);

    std::vector<double> finalTimeJob;
    std::vector<double> startTimeJob;

    std::vector<ValidationOp> certificate;

    auto start =
        std::chrono::steady_clock::now();

    double cost =
        TabuSearch(
            instance,
            parameters,
            machineCount,
            certificate,
            startTimeJob,
            bestMakespan,
            finalTimeJob);

    auto end =
        std::chrono::steady_clock::now();

    executionTime =
        std::chrono::duration<double>(
            end - start)
            .count();

    return cost;
}

// ============================================================
// Busca todas as 80 instâncias
// ============================================================

std::vector<fs::path> findInstances(
    const fs::path &dataDirectory)
{
    std::vector<fs::path> instances;

    for (const auto &entry :
         fs::directory_iterator(dataDirectory))
    {
        if (!entry.is_directory())
            continue;

        std::string name =
            entry.path()
                .filename()
                .string();

        if (name.rfind("I-", 0) != 0)
            continue;

        if (!isInstanceDirectory(
                entry.path()))
        {
            continue;
        }

        instances.push_back(
            entry.path());
    }

    std::sort(
        instances.begin(),
        instances.end());

    return instances;
}

// ============================================================
// MAIN
// ============================================================

int main(
    int argc,
    char *argv[])
{
    try
    {
        fs::path dataDirectory =
            (argc > 1)
                ? fs::path(argv[1])
                : fs::path("data");

        if (!fs::exists(dataDirectory))
        {
            std::cerr
                << "Diretorio nao encontrado: "
                << dataDirectory
                << '\n';

            return 1;
        }

        std::vector<fs::path> instances =
            findInstances(
                dataDirectory);

        if (instances.empty())
        {
            std::cerr
                << "Nenhuma instancia I-* encontrada em "
                << dataDirectory
                << '\n';

            return 1;
        }

        // Nome propositalmente diferente dos
        // experimentos anteriores.
        std::ofstream output(
            "critical_cplex_20s_results.csv");

        if (!output.is_open())
        {
            std::cerr
                << "Nao foi possivel criar "
                   "critical_cplex_20s_results.csv\n";

            return 1;
        }

        // =====================================
        // Cabeçalho do CSV
        // =====================================

        output
            << "Instance,"
            << "Critical_CPLEX_Cost,"
            << "Best_Makespan,"
            << "Time_Limit_Seconds,"
            << "Actual_Time_Seconds\n";

        output.flush();

        // =====================================
        // Informações do experimento
        // =====================================

        std::cout
            << "============================================\n"
            << "TESTE CRITICAL PATH + CPLEX - 20s\n"
            << "TABU - CRITICAL PATH + CPLEX TIMING\n"
            << "============================================\n"
            << "Instancias: "
            << instances.size()
            << "\n\n"
            << "Tempo por instancia: 20 s\n\n"
            << "Neighborhood: Critical Path (2)\n"
            << "Tabu tenure: 5\n"
            << "Cycle window: 6\n"
            << "Cycle repeats: 3\n"
            << "Checkpoint stack: 20\n"
            << "============================================\n\n";

        auto totalStart =
            std::chrono::steady_clock::now();

        // =====================================
        // Executa todas as instâncias
        // =====================================

        for (std::size_t i = 0;
             i < instances.size();
             ++i)
        {
            const fs::path &instanceDirectory =
                instances[i];

            std::string instanceName =
                instanceDirectory
                    .filename()
                    .string();

            long timeLimitMs =
                getTimeLimit(
                    instanceName);

            double timeLimitSeconds =
                timeLimitMs / 1000.0;

            std::cout
                << "["
                << (i + 1)
                << "/"
                << instances.size()
                << "] "
                << instanceName
                << " | limit="
                << timeLimitSeconds
                << " s ... "
                << std::flush;

            try
            {
                double bestMakespan =
                    0.0;

                double executionTime =
                    0.0;

                double cost =
                    runCritical(
                        instanceDirectory,
                        bestMakespan,
                        executionTime,
                        timeLimitMs);

                // =====================================
                // CSV
                // =====================================

                output
                    << instanceName
                    << ","
                    << std::fixed
                    << std::setprecision(6)
                    << cost
                    << ","
                    << bestMakespan
                    << ","
                    << timeLimitSeconds
                    << ","
                    << executionTime
                    << '\n';

                // Fundamental num teste de várias horas:
                // mesmo se der problema depois, tudo que
                // já terminou estará gravado em disco.
                output.flush();

                // =====================================
                // Terminal
                // =====================================

                std::cout
                    << "cost="
                    << std::fixed
                    << std::setprecision(2)
                    << cost
                    << " | makespan="
                    << bestMakespan
                    << " | time="
                    << std::setprecision(4)
                    << executionTime
                    << " s\n";
            }
            catch (const std::exception &error)
            {
                output
                    << instanceName
                    << ",ERROR,ERROR,"
                    << timeLimitSeconds
                    << ",ERROR\n";

                output.flush();

                std::cerr
                    << "\nERROR em "
                    << instanceName
                    << ": "
                    << error.what()
                    << '\n';
            }
            catch (...)
            {
                output
                    << instanceName
                    << ",ERROR,ERROR,"
                    << timeLimitSeconds
                    << ",ERROR\n";

                output.flush();

                std::cerr
                    << "\nERRO desconhecido em "
                    << instanceName
                    << '\n';
            }
        }

        output.close();

        // =====================================
        // Tempo total
        // =====================================

        auto totalEnd =
            std::chrono::steady_clock::now();

        double totalTime =
            std::chrono::duration<double>(
                totalEnd - totalStart)
                .count();

        std::cout
            << "\n============================================\n"
            << "TESTE FINALIZADO\n"
            << "============================================\n"
            << "Arquivo: critical_cplex_20s_results.csv\n"
            << "Tempo total: "
            << std::fixed
            << std::setprecision(2)
            << totalTime
            << " s\n"
            << "Tempo total: "
            << totalTime / 3600.0
            << " h\n"
            << "============================================\n";

        return 0;
    }
    catch (const std::exception &error)
    {
        std::cerr
            << "Erro fatal: "
            << error.what()
            << '\n';

        return 1;
    }
    catch (...)
    {
        std::cerr
            << "Erro fatal desconhecido.\n";

        return 1;
    }
}

#include "../include/ASAPResult.hpp"

#include <queue>

namespace
{
    std::vector<int> computeDegrees(const Instance &instance, int size)
    {
        std::vector<int> degree(size, 0);

        for (int i = 0; i < size; i++)
        {
            if (instance.predecessorJob[i] != -1)
                degree[i]++;

            if (instance.predecessorMach[i] != -1)
                degree[i]++;
        }

        return degree;
    }
    std::queue<int> buildInitialReadyQueue(const std::vector<int> &degree, int size)
    {
        std::queue<int> readyQueue;

        for (int i = 0; i < size; i++)
        {
            if (degree[i] == 0)
                readyQueue.push(i);
        }

        return readyQueue;
    }

    double computeSetupTime(const Instance &instance, int current)
    {
        int machine = instance.operationsList[current].id_machine;
        int job = instance.operationsList[current].id_job;
        int predecessorMachine = instance.predecessorMach[current];

        if (predecessorMachine != -1)
        {
            int previousJob = instance.operationsList[predecessorMachine].id_job;
            return instance.setupMatrix[machine][previousJob][job];
        }

        int initialJob = instance.initialSetup[machine];
        if (initialJob == -1 || initialJob == job)
            return 0.0;

        return instance.setupMatrix[machine][initialJob][job];
    }

    double computeJobReadyTime(const Instance &instance, std::vector<double> &finalTimeOperation, int current)
    {
        int predecessorJob = instance.predecessorJob[current];

        if (predecessorJob == -1)
            return instance.jobsList[instance.operationsList[current].id_job].release_date;

        return finalTimeOperation[predecessorJob];
    }

    double computeMachineReadyTime(const Instance &instance, std::vector<double> &finalTimeOperation, int current,
                                   double setupTime)
    {
        int predecessorMachine = instance.predecessorMach[current];

        if (predecessorMachine == -1)
            return setupTime;

        return finalTimeOperation[predecessorMachine] + setupTime;
    }

    double selectBottleneck(double timeJobFinal, double timeMachineFinal, int predecessorJob, int predecessorMach,
                            int &criticalPredecessorOut)
    {
        if (timeJobFinal > timeMachineFinal)
        {
            criticalPredecessorOut = predecessorJob;

            return timeJobFinal;
        }

        if (timeMachineFinal > timeJobFinal)
        {
            criticalPredecessorOut = predecessorMach;

            return timeMachineFinal;
        }

        if (predecessorMach != -1)
        {
            criticalPredecessorOut = predecessorMach;
        }
        else
        {
            criticalPredecessorOut = predecessorJob;
        }

        return timeJobFinal;
    }
    void releaseSuccessorJob(std::queue<int> &readyQueue, std::vector<int> &degree, const Instance &instance,
                             int current, bool &isLastOperation)
    {
        int successor = instance.successorJob[current];

        if (successor == -1)
        {
            isLastOperation = true;
            return;
        }

        degree[successor]--;

        if (degree[successor] == 0)
            readyQueue.push(successor);
    }

    void releaseSuccessorMachine(std::queue<int> &readyQueue, std::vector<int> &degree, const Instance &instance, int current)
    {
        int successor = instance.successorMach[current];

        if (successor == -1)
            return;

        degree[successor]--;

        if (degree[successor] == 0)
            readyQueue.push(successor);
    }

    double findMakespan(const std::vector<double> &jobsCost)
    {
        double makespan = 0.0;

        for (double time : jobsCost)
        {
            if (time > makespan)
                makespan = time;
        }

        return makespan;
    }
}
double penaltySum(const std::vector<double> &finalTimeJob, const Instance &instance,
                  const std::vector<double> &startTimeJob)
{
    double sum = 0;
    for (size_t i = 0; i < finalTimeJob.size(); i++)
    {
        double due_date = instance.jobsList[i].due_date;
        double C_last = finalTimeJob[i];
        double S_first = startTimeJob[i];
        double flow_time = C_last - S_first;

        if (C_last < due_date)
            sum += (due_date - C_last) * instance.jobsList[i].earliness_penalty;
        else if (C_last > due_date)
            sum += (C_last - due_date) * instance.jobsList[i].tardiness_penalty;

        sum += flow_time * instance.jobsList[i].flow_time;
    }
    return sum;
}
ASAPResult calculateASAP(const Instance &instance)
{
    ASAPResult result;

    double startTimeOperation;
    double finalTimeJob;
    double finalTimeMachine;

    int size = instance.operationsList.size();
    int jobsCount = instance.jobsList.size();
    int processed = 0;
    int currentJob;

    std::vector<double> startTimeJob(jobsCount, 0.0);

    result.finalTimeJob.resize(jobsCount, 0.0);

    std::vector<int> degree = computeDegrees(instance, size);

    std::queue<int> readyQueue = buildInitialReadyQueue(degree, size);

    result.finalTimeOperation.resize(size, 0.0);
    result.criticalPredecessor.resize(size, -1);

    while (!readyQueue.empty())
    {
        int current = readyQueue.front();
        readyQueue.pop();

        double setupTime = computeSetupTime(instance, current);

        int criticalPredecessorOut = -1;

        finalTimeJob = computeJobReadyTime(instance, result.finalTimeOperation, current);

        finalTimeMachine = computeMachineReadyTime(instance, result.finalTimeOperation, current, setupTime);

        startTimeOperation = selectBottleneck(finalTimeJob, finalTimeMachine, instance.predecessorJob[current],
                                              instance.predecessorMach[current], criticalPredecessorOut);

        result.criticalPredecessor[current] = criticalPredecessorOut;

        currentJob = instance.operationsList[current].id_job;

        if (instance.predecessorJob[current] == -1)
        {
            startTimeJob[currentJob] = startTimeOperation;
        }

        double finalTimeOperation = startTimeOperation + instance.processingTime[current];

        result.finalTimeOperation[current] = finalTimeOperation;

        bool isLastOperation = false;

        releaseSuccessorJob(readyQueue, degree, instance, current, isLastOperation);

        releaseSuccessorMachine(readyQueue, degree, instance, current);

        if (isLastOperation)
        {
            result.finalTimeJob[currentJob] = finalTimeOperation;
        }

        processed++;
    }

       if (processed != size)
    {
        result.feasible = false;
        return result;
    }

    result.feasible = true;

    result.makespan = findMakespan(result.finalTimeJob);

    result.cost = penaltySum(result.finalTimeJob, instance, startTimeJob);

    return result;
}

void CheckpointStack::push(SearchCheckpoint &checkpoint)
{
    if (checkpoint.remainingMoves.empty())
        return;

    checkpoint.isEmpty = false;

    if (empty)
    {
        empty = false;
        base = 0;
        top = 0;

        stack[0] = checkpoint;

        return;
    }

    top = (top + 1) % maxSize;

    if (top == base)
        base = (base + 1) % maxSize;

    stack[top] = checkpoint;
}
SearchCheckpoint CheckpointStack::pop()
{
    if (empty)
    {
        SearchCheckpoint checkpoint;
        return checkpoint;
    }

    while (stack[top].remainingMoves.empty())
    {
        if (top == base)
        {
            empty = true;

            SearchCheckpoint checkpoint;
            return checkpoint;
        }
        else
            top = (top + maxSize - 1) % maxSize;
    }

    return stack[top];
}
void CheckpointStack::updateRemainingMoves(const std::vector<Movement> &remaining)
{
    if (empty)
        return;

    stack[top].remainingMoves = remaining;

    while (stack[top].remainingMoves.empty())
    {
        stack[top].isEmpty = true;

        if (base == top)
        {
            empty = true;
            return;
        }
        else
            top = (top + maxSize - 1) % maxSize;
    }
}

{
    std::ifstream file(path);

    if (!file.is_open())
        throw std::runtime_error("Could not open file: " + path);

    std::vector<std::vector<std::string>> rows;
    std::string line;

    if (skipHeader)
        std::getline(file, line);

    while (std::getline(file, line))
    {
        if (line.empty())
            continue;

        rows.push_back(splitLine(line));
    }

    return rows;
}

CycleDetector::CycleDetector(unsigned cycleWindow, unsigned cycleRepeats)
{
    costHistory.resize(cycleWindow, -1);
    window = cycleWindow;
    requiredRepeats = cycleRepeats;
}
//(ex: 10, 12, 15, 10, 12, 15, 10, 12, 15, ...) d
bool CycleDetector::detect(double currentCost, bool isNewBest)
{
    if (isNewBest)
    {
        writePointer = 0;
        suspectedPeriod = 0;
        repeatsSoFar = 0;
        for (size_t i = 0; i < costHistory.size(); i++)
            costHistory[i] = -1;
    }

    double tolerance = 0.10;
    if (suspectedPeriod != 0)
    {
        size_t index = (writePointer + window - suspectedPeriod) % window;
        if (std::abs(currentCost - costHistory[index]) <= tolerance)
        {
            repeatsSoFar++;

            if (repeatsSoFar >= requiredRepeats)
            {
                costHistory[writePointer] = currentCost;
                ++writePointer;
                writePointer = writePointer % window;
                return true;
            }
        }
        else
        {
            suspectedPeriod = 0;
            repeatsSoFar = 0;

            unsigned int count = 1;

            while (count < window)
            {
                int index = (writePointer + window - count) % window;
                if (std::abs(currentCost - costHistory[index]) <= tolerance)
                    suspectedPeriod = count;
                count++;
            }
        }
    }
    else
    {
        unsigned int count = 1;

        while (count < window)
        {
            int index = (writePointer + window - count) % window;
            if (std::abs(currentCost - costHistory[index]) <= tolerance)
                suspectedPeriod = count;
            count++;
        }
    }

    costHistory[writePointer] = currentCost;
    ++writePointer;
    writePointer = writePointer % window;

    return false;
}

void createSPT(Instance &instance, Solution &solution, int n_mach)
{
    solution.solution_matrix.clear();
    solution.solution_matrix.resize(n_mach);
    int size_op = instance.operationsList.size();
    int size_job = instance.jobsList.size();
    int processed = 0;
    std::vector<int> job_processing;
    job_processing.resize(size_job, 0);

    while (processed < size_op)
    {
        int best_op = -1;
        int best_job = -1;
        double lower_time = 999999999.0;

        for (int i = 0; i < size_job; i++)
        {
            if (job_processing[i] < (int)instance.jobOperation[i].size())
            {
                int op_index = instance.jobOperation[i][job_processing[i]];
                double op_time = instance.operationsList[op_index].processing_time;

                if (op_time < lower_time)
                {
                    best_job = i;
                    best_op = op_index;
                    lower_time = op_time;
                }
            }
        }

        if (best_op != -1)
        {
            solution.solution_matrix[instance.operationsList[best_op].id_machine].push_back(best_op);
            job_processing[best_job]++;
            processed++;
        }
        else
        {
            break; // Se não achar ninguém, sai do loop pra não travar o PC
        }
    }
}

std::vector<Movement> adjacentNeighborhood(Solution &current)
{
    std::vector<Movement> neighbors;
    for (size_t mach = 0; mach < current.solution_matrix.size(); mach++)
    {
        if (current.solution_matrix[mach].size() < 2)
            continue;

        for (size_t i = 0; i < current.solution_matrix[mach].size() - 1; i++)
        {
            Movement mov;
            mov.solution = current;
            mov.op1 = mov.solution.solution_matrix[mach][i];
            mov.op2 = mov.solution.solution_matrix[mach][i + 1];

            std::swap(mov.solution.solution_matrix[mach][i], mov.solution.solution_matrix[mach][i + 1]);

            neighbors.push_back(mov);
        }
    }
    return neighbors;
}
std::vector<int> generateCriticalPath(Instance &instance)
{
    ASAPResult result = calculateASAP(instance);

    std::vector<int> criticalPath;

    if (!result.feasible || result.finalTimeJob.empty())
    {
        return criticalPath;
    }

    int index_last_job = -1;

    for (size_t i = 0; i < result.finalTimeJob.size(); i++)
    {
        if (std::abs(result.finalTimeJob[i] - result.makespan) < 1e-9)
        {
            index_last_job = i;
        }
    }

    int current = instance.jobOperation[index_last_job].back();

    while (current != -1)
    {
        criticalPath.push_back(current);
        current = result.criticalPredecessor[current];
    }

    return criticalPath;
}

std::vector<Movement> criticalPathNeighborhood(Solution &current, std::vector<int> &criticalPath, Instance &instance)
{
    std::vector<Movement> neighbors;

    if (criticalPath.empty())
    {
        throw std::runtime_error("Critical Path is empty");
    }

    for (size_t j = 0; j < criticalPath.size() - 1; j++)
    {
        int op1 = criticalPath[j];
        int op2 = criticalPath[j + 1];

        int mach1 = instance.operToMach[op1];
        int mach2 = instance.operToMach[op2];

        if (mach1 == mach2)
        {
            Movement mov;
            mov.op1 = op1;
            mov.op2 = op2;
            mov.solution = current;

            int index_op1 = -1;
            int index_op2 = -1;

            for (size_t i = 0; i < mov.solution.solution_matrix[mach1].size(); i++)
            {
                if (mov.solution.solution_matrix[mach1][i] == op1)
                    index_op1 = i;

                if (mov.solution.solution_matrix[mach1][i] == op2)
                    index_op2 = i;
            }

            if (index_op1 != -1 && index_op2 != -1 && std::abs(index_op1 - index_op2) == 1)
            {
                std::swap(mov.solution.solution_matrix[mach1][index_op1], mov.solution.solution_matrix[mach1][index_op2]);
                neighbors.push_back(mov);
            }
        }
    }
    return neighbors;
}

bool searchTimeExpired(
    const std::chrono::steady_clock::time_point &startTime,
    long maxMilliSeconds)
{
    // -1 significa sem limite de tempo
    if (maxMilliSeconds < 0)
        return false;

    auto now = std::chrono::steady_clock::now();

    auto elapsed =
        std::chrono::duration_cast<std::chrono::milliseconds>(
            now - startTime)
            .count();

    return elapsed >= maxMilliSeconds;

void changePredecessorMach(const Solution &solution, Instance &instance)
{
    for (const auto &fila : solution.solution_matrix)
    {

        instance.predecessorMach[fila[0]] = -1;

        for (size_t i = 0; i < fila.size() - 1; i++)
        {
            instance.successorMach[fila[i]] = fila[i + 1];
        }
    }

double computeSetupTime(const Instance &instance, int current)
{
    int machine = instance.operationsList[current].id_machine;
    int job = instance.operationsList[current].id_job;
    int predecessorMachine = instance.predecessorMach[current];

    if (predecessorMachine != -1)
    {
        int previousJob = instance.operationsList[predecessorMachine].id_job;
        return instance.setupMatrix[machine][previousJob][job];
    }

    int initialJob = instance.initialSetup[machine];
    if (initialJob == -1 || initialJob == job)
        return 0.0;

    return instance.setupMatrix[machine][initialJob][job];
}

SolutionState evaluateASAPSolution(const Solution &solution, Instance &instance)
{
    evaluationCount++;

    SolutionState state;

    state.solution = solution;

    changeMachines(solution, instance);

    ASAPResult asap =
        calculateASAP(instance);

    if (!asap.feasible)
        return state;

    state.feasible = true;

    state.cost = asap.cost;

    return state;
}
SolutionState evaluateSolution(const Solution &solution, Instance &instance, TimingSolver &solver)
{
    cplexCount++;
    SolutionState state;

    state.solution = solution;

    changeMachines(solution, instance);

    bool isFeasible = valuer(instance);

    if (!isFeasible)
        return state;

    state.feasible = true;

    CostResult timing = solver.solveCost(state, instance);

    if (!timing.feasible)
    {
        state.feasible = false;
        return state;
    }

    state.cost = timing.cost;

    state.timingOptimal = timing.optimal;

    return state;
}

bool chooseBestCandidate(const std::vector<Movement> &neighborhood, TabuList &tabuList, Instance &instance,
                         int &tabuIteration, int &chosenIndex, int &chosenOp1, int &chosenOp2,
                         SolutionState &bestCandidate, const std::chrono::steady_clock::time_point &searchStart,
                         long maxMilliSeconds, bool &timeExpired, TimingSolver &solver)
{
    double infinity = std::numeric_limits<double>::infinity();

    double bestASAPCost = infinity;

    bool movementFound = false;

    chosenIndex = -1;
    chosenOp1 = -1;
    chosenOp2 = -1;

    Solution bestASAPSolution;

    for (size_t i = 0; i < neighborhood.size(); i++)
    {
        if (searchTimeExpired(searchStart, maxMilliSeconds))
        {
            timeExpired = true;
            return false;
        }

        const Solution &candidate = neighborhood[i].solution;

        int op1 = neighborhood[i].op1;

        int op2 = neighborhood[i].op2;

        // Primeiro olha tabu.
        bool isTabu = tabuList.isTabu(op1, op2, tabuIteration);

        if (isTabu)
            continue;

        SolutionState stateCandidate = evaluateASAPSolution(candidate, instance);

        if (!stateCandidate.feasible)
            continue;

        if (stateCandidate.cost < bestASAPCost)
        {
            bestASAPCost = stateCandidate.cost;

            bestASAPSolution = candidate;

            chosenIndex = static_cast<int>(i);

            chosenOp1 = op1;
            chosenOp2 = op2;

            movementFound = true;
        }
    }

    if (!movementFound)
        return false;

    if (searchTimeExpired(searchStart, maxMilliSeconds))
    {
        timeExpired = true;
        return false;
    }

    bestCandidate = evaluateSolution(bestASAPSolution, instance, solver);

    if (searchTimeExpired(searchStart, maxMilliSeconds))
    {
        timeExpired = true;
        return false;
    }

    if (!bestCandidate.feasible)
        return false;

    return true;
}

bool restoreCheckpoint(CheckpointStack &checkpointStack, Instance &instance, SolutionState &current, TabuList &tabuList,
                       int &tabuIteration, std::vector<Movement> &neighborhood, TimingSolver &solver)
{
    SearchCheckpoint checkpoint = checkpointStack.pop();

    if (checkpoint.isEmpty)
        return false;

    current = evaluateSolution(checkpoint.solution, instance, solver);
    tabuList = checkpoint.tabuListSnapshot;
    tabuIteration = checkpoint.tabuIterationSnapshot;

    neighborhood = checkpoint.remainingMoves;

    return true;
}

std::vector<Movement> remainingAfterChoice(const std::vector<Movement> &neighborhood, int chosenIndex)
{
    std::vector<Movement> remaining = neighborhood;
    remaining.erase(remaining.begin() + chosenIndex);

    return remaining;
}

void checkpointAfterChoice(CheckpointStack &checkpointStack, const SolutionState &current, const TabuList &tabuList,
                           int tabuIteration, const std::vector<Movement> &remaining)
{
    SearchCheckpoint newCheckpoint;

    newCheckpoint.solution = current.solution;
    newCheckpoint.remainingMoves = remaining;
    newCheckpoint.tabuListSnapshot = tabuList;
    newCheckpoint.tabuIterationSnapshot = tabuIteration;

    checkpointStack.push(newCheckpoint);
    return;
}

void updateBest(const SolutionState &current, SolutionState &best, int &iterationsWithoutImprovement, bool &isNewBest)
{
    if (current.cost < best.cost)
    {
        best = current;
        iterationsWithoutImprovement = 0;
        isNewBest = true;
    }
    else
    {
        iterationsWithoutImprovement++;
        isNewBest = false;
    }
}

std::vector<ValidationOp> buildCertificate(const Instance &instance, const TimingResult &timing)
{

    size_t size = instance.operationsList.size();
    std::vector<ValidationOp> certificate;
    certificate.resize(size);

    for (size_t i = 0; i < size; i++)
    {
        ValidationOp entry;
        int current = instance.operationsList[i].id_operation;

        entry.id_operation = current;
        entry.id_job = instance.operationsList[current].id_job;
        entry.id_machine = instance.operationsList[current].id_machine;
        entry.release_date = instance.jobsList[instance.operationsList[current].id_job].release_date;
        entry.start_time = timing.startTimeOp[current];
        entry.final_time = timing.finalTimeOp[current];
        entry.processing_time = instance.processingTime[current];
        entry.setup_time = computeSetupTime(instance, current);

        certificate[current] = entry;
    }
    return certificate;
}

double TabuSearch(Instance &instance, Parameters &parameters, int machineCount, std::vector<ValidationOp> &certificate,
                  std::vector<double> &startTimeJob, double &bestMakespan, std::vector<double> &finalTimeJob)
{
    evaluationCount = 0;
    auto searchStart = std::chrono::steady_clock::now();

    TimingSolver solver(instance);

    Solution initialSolution;

    createSPT(instance, initialSolution, machineCount);

    SolutionState current = evaluateSolution(initialSolution, instance, solver);

    SolutionState best = current;

    int tabuSize = instance.operationsList.size();

    TabuList tabuList(tabuSize);

    CycleDetector cycleDetector(parameters.cycleWindow, parameters.cycleRepeats);

    CheckpointStack checkpointStack(parameters.checkpointStackSize);

    bool isNewBest = true;

    int currentIteration = 0;
    int tabuIteration = 0;

    int iterationsWithoutImprovement = 0;

    bool stopByTime = false;

    while (iterationsWithoutImprovement < parameters.iterationsWithoutImprovement)
    {
        if (searchTimeExpired(searchStart, parameters.maxMilliSeconds))
        {
            break;
        }

        currentIteration++;
        tabuIteration++;

        bool cycling = cycleDetector.detect(current.cost, isNewBest);

        bool jumped;

        std::vector<Movement> neighborhood;

        if (cycling)
        {
            bool restored = restoreCheckpoint(checkpointStack, instance, current, tabuList, tabuIteration, neighborhood, solver);

            if (!restored)
                break;

            jumped = true;
        }
        else
        {
            neighborhood = chosenNeighborhood(parameters.neighborhood, current.solution, instance);

            jumped = false;
        }

        if (neighborhood.empty())
        {
            bool restored = restoreCheckpoint(checkpointStack, instance, current, tabuList, tabuIteration, neighborhood, solver);

            if (!restored)
                break;

            jumped = true;
        }

        int chosenIndex;
        int chosenOp1;
        int chosenOp2;

        SolutionState bestCandidate;

        bool timeExpired = false;

        bool movementFound = chooseBestCandidate(neighborhood, tabuList, instance, tabuIteration, chosenIndex, chosenOp1,
                                                 chosenOp2, bestCandidate, searchStart, parameters.maxMilliSeconds, timeExpired,
                                                 solver);

        if (timeExpired)
        {
            stopByTime = true;
            break;
        }

        bool terminateSearch = false;

        while (!movementFound)
        {
            // Também precisamos verificar o
            // tempo dentro deste segundo loop.
            if (searchTimeExpired(searchStart, parameters.maxMilliSeconds))
            {
                stopByTime = true;
                break;
            }

            if (jumped)
            {
                std::vector<Movement> emptyRemaining;

                checkpointStack.updateRemainingMoves(emptyRemaining);
            }

            bool restored = restoreCheckpoint(checkpointStack, instance, current, tabuList, tabuIteration, neighborhood,
                                              solver);

            if (!restored)
            {
                terminateSearch = true;
                break;
            }

            timeExpired = false;

            movementFound = chooseBestCandidate(neighborhood, tabuList, instance, tabuIteration, chosenIndex,
                                                chosenOp1, chosenOp2, bestCandidate, searchStart,
                                                parameters.maxMilliSeconds, timeExpired, solver);

            if (timeExpired)
            {
                stopByTime = true;
                break;
            }

            jumped = true;
        }

        if (stopByTime)
            break;

        if (terminateSearch)
            break;

        std::vector<Movement> remaining = remainingAfterChoice(neighborhood, chosenIndex);

        if (isNewBest && !jumped)
        {
            checkpointAfterChoice(checkpointStack, current, tabuList, tabuIteration, remaining);
        }

        if (jumped)
        {
            checkpointStack.updateRemainingMoves(remaining);
        }

        current = bestCandidate;

        changeMachines(current.solution, instance);

        tabuList.insertMovement(chosenOp1, chosenOp2, tabuIteration, parameters.tabuTenure);

        tabuList.insertMovement(chosenOp2, chosenOp1, tabuIteration, parameters.tabuTenure);

        updateBest(current, best, iterationsWithoutImprovement, isNewBest);
    }

    // Mesmo que tenha parado pelo tempo,
    // devolvemos o MELHOR encontrado até então.

    changeMachines(best.solution, instance);

    TimingResult finalTiming = solver.solveFull(best, instance);

    if (!finalTiming.feasible)
    {
        std::cerr << "Erro ao temporizar a melhor solucao final." << std::endl;

        return std::numeric_limits<double>::infinity();
    }

    startTimeJob = finalTiming.startTimeJob;

    finalTimeJob = finalTiming.finalTimeJob;

    bestMakespan = finalTiming.makespan;

    std::cout << "Evaluations: " << evaluationCount << std::endl;
    std::cout << "CPLEX Calls: " << cplexCount << std::endl;

    certificate = buildCertificate(instance, finalTiming);

    return best.cost;
}

#include "../include/timingSolver.hpp"
#include "../include/tabuSearch.hpp"

#include <limits>
#include <iostream>

TimingSolver::TimingSolver(const Instance &instance)
    : model(env),
      S(env, instance.operationsList.size(), 0.0, IloInfinity, ILOFLOAT),
      earliness(env, instance.jobsList.size(), 0.0, IloInfinity, ILOFLOAT),
      tardiness(env, instance.jobsList.size(), 0.0, IloInfinity, ILOFLOAT),
      objective(env),
      cplex(env),
      machineConstraints(env)
{
    size_t sizeJob = instance.jobsList.size();

    lastOp.resize(sizeJob, -1);
    firstOp.resize(sizeJob, -1);

    for (size_t i = 0; i < sizeJob; i++)
    {
        if (instance.jobOperation[i].empty())
        {
            error = true;
            return;
        }

        firstOp[i] = instance.jobOperation[i].front();

        lastOp[i] = instance.jobOperation[i].back();
    }

    addJobPrecedenceConstraints(model, S, instance);
    addEarlinessTardinessConstraints(model, S, instance, lastOp, earliness, tardiness);
    buildObjectiveExpression(instance, earliness, tardiness, objective, S, lastOp, firstOp);

    model.add(IloMinimize(env, objective));

    cplex.setParam(IloCplex::Param::Threads, 1);
    // cplex.setParam(IloCplex::Param::TimeLimit, 1);

    cplex.setOut(env.getNullStream());
    cplex.setWarning(env.getNullStream());

    cplex.extract(model);
}

TimingSolver::~TimingSolver()
{
    env.end();
}

void TimingSolver::clearMachineConstraints()
{
    if (machineConstraints.getSize() == 0)
        return;

    model.remove(machineConstraints);

    machineConstraints.endElements();

    machineConstraints.clear();
}
void TimingSolver::addJobPrecedenceConstraints(IloModel &model, IloNumVarArray &S, const Instance &instance)
{
    size_t sizeOp = instance.operationsList.size();

    for (size_t i = 0; i < sizeOp; i++)
    {
        if (instance.predecessorJob[i] == -1)
        {
            int indexOp = instance.operationsList[i].id_job;
            model.add(S[i] >= instance.jobsList[indexOp].release_date);
        }
        else
        {
            int predecessor = instance.predecessorJob[i];
            model.add(S[i] >= S[predecessor] + instance.processingTime[predecessor]);
        }
    }
}

void TimingSolver::addMachinePrecedenceConstraints(const Instance &instance)
{
    size_t sizeOp = instance.operationsList.size();

    for (size_t i = 0; i < sizeOp; i++)
    {
        int machine = instance.operationsList[i].id_machine;
        int job = instance.operationsList[i].id_job;

        double st = -1;
        if (instance.predecessorMach[i] == -1)
        {
            int initialJob = instance.initialSetup[machine];

            if (initialJob != -1 && initialJob != job)
            {
                st = instance.setupMatrix[machine][initialJob][job];

                IloConstraint constraint = S[i] >= st;

                machineConstraints.add(constraint);
            }
        }

        if (instance.predecessorMach[i] != -1)
        {
            int predecessorOp = instance.predecessorMach[i];
            int predecessor = instance.operationsList[predecessorOp].id_job;
            st = instance.setupMatrix[machine][predecessor][job];
            IloConstraint constraint = S[i] >= S[predecessorOp] + instance.processingTime[predecessorOp] + st;

            machineConstraints.add(constraint);
        }
    }

    model.add(machineConstraints);
}

TimingResult makeFailedTimingResult()
{
    TimingResult failed;

    failed.feasible = false;
    failed.optimal = false;
    failed.cost = std::numeric_limits<double>::infinity();
    failed.makespan = std::numeric_limits<double>::infinity();

    return failed;
}

CostResult makeFailedCostResult()
{
    CostResult failed;

    failed.feasible = false;
    failed.optimal = false;
    failed.cost = std::numeric_limits<double>::infinity();

    return failed;
}

void TimingSolver::addEarlinessTardinessConstraints(IloModel &model, IloNumVarArray &S, const Instance &instance,
                                                    std::vector<int> &lastOp, IloNumVarArray &earliness, IloNumVarArray &tardiness)
{
    size_t sizeJob = instance.jobsList.size();

    for (size_t i = 0; i < sizeJob; i++)
    {
        int last = lastOp[i];
        double dueDate = instance.jobsList[i].due_date;
        model.add(dueDate == (S[last] + instance.processingTime[last]) + earliness[i] - tardiness[i]);
    }
}

void TimingSolver::buildObjectiveExpression(const Instance &instance, IloNumVarArray &earliness, IloNumVarArray &tardiness,
                                            IloExpr &objective, IloNumVarArray &S, std::vector<int> &lastOp, std::vector<int> &firstOp)
{
    size_t sizeJob = instance.jobsList.size();

    for (size_t i = 0; i < sizeJob; i++)
    {
        double weight_e, weight_t, weight_ft;
        weight_e = instance.jobsList[i].earliness_penalty;
        weight_t = instance.jobsList[i].tardiness_penalty;
        weight_ft = instance.jobsList[i].flow_time;

        int last = lastOp[i];
        int first = firstOp[i];
        double pt = instance.processingTime[last];
        objective += ((weight_e * earliness[i]) + (weight_t * tardiness[i]) + (weight_ft * ((S[last] + pt) - S[first])));
    }
}

void extractTimingResult(IloNumVarArray &S, const Instance &instance, TimingResult &result, IloCplex &cplex,
                         std::vector<int> &lastOp, std::vector<int> &firstOp)
{
    size_t sizeJob = instance.jobsList.size();
    size_t sizeOp = instance.operationsList.size();

    result.startTimeOp.resize(sizeOp, 0);
    result.finalTimeOp.resize(sizeOp, 0);
    result.finalTimeJob.resize(sizeJob, 0);
    result.startTimeJob.resize(sizeJob, 0);

    for (size_t i = 0; i < sizeOp; i++)
    {
        result.startTimeOp[i] = cplex.getValue(S[i]);
        result.finalTimeOp[i] = result.startTimeOp[i] + instance.processingTime[i];
    }

    for (size_t i = 0; i < sizeJob; i++)
    {
        int first = firstOp[i];
        int last = lastOp[i];

        result.startTimeJob[i] = result.startTimeOp[first];

        result.finalTimeJob[i] = result.finalTimeOp[last];
    }

    double makespan = 0.0;

    for (size_t i = 0; i < sizeJob; i++)
    {
        if (result.finalTimeJob[i] > makespan)
            makespan = result.finalTimeJob[i];
    }

    result.makespan = makespan;
    result.cost = cplex.getObjValue();
    result.feasible = true;
}

CostResult TimingSolver::solveCost(
    const SolutionState &state,
    Instance &instance)
{
    if (!state.feasible)
    {
        return makeFailedCostResult();
    }

    if (error)
    {
        return makeFailedCostResult();
    }

    CostResult result;

    try
    {
        clearMachineConstraints();

        addMachinePrecedenceConstraints(
            instance);

        bool solved =
            cplex.solve();

        if (!solved)
        {
            return makeFailedCostResult();
        }

        result.feasible = true;

        result.optimal =
            cplex.getStatus() ==
            IloAlgorithm::Optimal;

        result.cost =
            cplex.getObjValue();
    }
    catch (IloException &e)
    {
        std::cerr
            << "Concert/CPLEX Error: "
            << e
            << std::endl;

        return makeFailedCostResult();
    }
    catch (const std::exception &e)
    {
        std::cerr
            << "C++ Error: "
            << e.what()
            << std::endl;

        return makeFailedCostResult();
    }
    catch (...)
    {
        std::cerr
            << "Erro desconhecido no TimingSolver."
            << std::endl;

        return makeFailedCostResult();
    }

    return result;
}

TimingResult TimingSolver::solveFull(const SolutionState &solutionState, Instance &instance)
{
    if (!solutionState.feasible)
    {
        return makeFailedTimingResult();
    }

    if (error)
    {
        return makeFailedTimingResult();
    }

    TimingResult result;

    try
    {
        clearMachineConstraints();

        addMachinePrecedenceConstraints(instance);

        bool solved = cplex.solve();

        if (!solved)
        {
            result = makeFailedTimingResult();
        }
        else
        {
            IloAlgorithm::Status status = cplex.getStatus();

            if (status == IloAlgorithm::Optimal)
            {
                result.optimal = true;
            }
            else
            {
                result.optimal = false;
                std::cout << "O resultado não é ótimo! Status: "
                          << status << std::endl;
            }
            extractTimingResult(S, instance, result, cplex, lastOp, firstOp);
            // cplex.exportModel("timing.lp");
        }
    }
    catch (IloException &e)
    {
        std::cerr << "Concert/CPLEX Error: " << e << std::endl;
        result = makeFailedTimingResult();
    }
    catch (const std::exception &e)
    {
        std::cerr << "C++ Error: "
                  << e.what() << std::endl;

        result = makeFailedTimingResult();
    }
    catch (...)
    {
        std::cerr << "Erro desconhecido no TimingSolver."
                  << std::endl;

        result = makeFailedTimingResult();
    }

    return result;
}

#include "../include/instance.hpp"
#include <vector>
#include <queue>

std::vector<int> computeDegrees(const Instance &instance, int size)
{
    std::vector<int> degree(size, 0);

    for (int i = 0; i < size; i++)
    {
        if (instance.predecessorJob[i] != -1)
            degree[i]++;

        if (instance.predecessorMach[i] != -1)
            degree[i]++;
    }

    return degree;
}
std::queue<int> buildInitialReadyQueue(const std::vector<int> &degree, int size)
{
    std::queue<int> readyQueue;

    for (int i = 0; i < size; i++)
    {
        if (degree[i] == 0)
            readyQueue.push(i);
    }

    return readyQueue;
}

void releaseSuccessorJob(std::queue<int> &readyQueue, std::vector<int> &degree, const Instance &instance, int current)
{
    int successor = instance.successorJob[current];
    if (successor == -1)
        return;

    degree[successor]--;

    if (degree[successor] == 0)
        readyQueue.push(successor);
}

void releaseSuccessorMachine(std::queue<int> &readyQueue, std::vector<int> &degree, const Instance &instance, int current)
{
    int successor = instance.successorMach[current];

    if (successor == -1)
        return;

    degree[successor]--;

    if (degree[successor] == 0)
        readyQueue.push(successor);
}

bool valuer(const Instance &instance)
{
    int size = instance.operationsList.size();
    int processed = 0;

    std::vector<int> degree = computeDegrees(instance, size);
    std::queue<int> readyQueue = buildInitialReadyQueue(degree, size);

    while (!readyQueue.empty())
    {
        int current = readyQueue.front();
        readyQueue.pop();

        releaseSuccessorJob(readyQueue, degree, instance, current);
        releaseSuccessorMachine(readyQueue, degree, instance, current);

        processed++;
    }

    if (processed != size)
        return false;

    return true;
}
